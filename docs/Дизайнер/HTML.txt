function getUsers() {
  try { return JSON.parse(localStorage.getItem('timeRoomUsers') || '[]'); }
  catch { return []; }
}
function saveUsers(users) { localStorage.setItem('timeRoomUsers', JSON.stringify(users)); }
function findUser(fio, phone) {
  const users = getUsers();
  const fioNorm = fio.trim().toLowerCase();
  const phoneNorm = phone.replace(/\D/g, '');
  return users.find(u => u.fio.toLowerCase() === fioNorm && u.phone.replace(/\D/g, '') === phoneNorm);
}
function updateAuthButton() {
  const btn = document.getElementById('authBtn');
  if (currentUser) {
    btn.textContent = currentUser.fio;
    btn.classList.add('logged-in');
    btn.onclick = () => {
      if (confirm('Выйти из аккаунта?')) {
        currentUser = null;
        updateAuthButton();
      }
    };
  } else {
    btn.textContent = 'Войти / Зарегистрироваться';
    btn.classList.remove('logged-in');
    btn.onclick = openAuthModal;
  }
}
function openAuthModal() {
  document.getElementById('authFio').value = '';
  document.getElementById('authPhone').value = '';
  document.getElementById('authModal').classList.add('active');
}
function validateAuthFields() {
  const fio = document.getElementById('authFio').value.trim();
  const phone = document.getElementById('authPhone').value.trim();
  const fioValid = fio.split(/\s+/).length >= 2 && fio.length >= 5;
  const phoneDigits = phone.replace(/\D/g, '');
  const phoneValid = phoneDigits.length >= 10;
  return { fio, phone, fioValid, phoneValid, valid: fioValid && phoneValid };
}
function showWarning(text) {
  document.getElementById('warningText').textContent = text;
  document.getElementById('warningModal').classList.add('active');
}
function handleRegister() {
  const { fio, phone, valid } = validateAuthFields();
  if (!valid) { showWarning('Вы ввели не все данные или они не корректны'); return; }
  const existing = findUser(fio, phone);
  if (existing) { showWarning('Вы уже зарегистрированы! Пожалуйста, нажмите кнопку Войти'); return; }
  const users = getUsers();
  users.push({ fio, phone });
  saveUsers(users);
  currentUser = { fio, phone };
  updateAuthButton();
  closeModal('authModal');
  alert('Регистрация успешна! Добро пожаловать, ' + fio);
}
function handleLogin() {
  const { fio, phone, valid } = validateAuthFields();
  if (!valid) { showWarning('Вы ввели не все данные или они не корректны'); return; }
  const existing = findUser(fio, phone);
  if (!existing) { showWarning('Вы еще не зарегистрированы! Пожалуйста, нажмите кнопку Зарегистрироваться'); return; }
  currentUser = { fio, phone };
  updateAuthButton();
  closeModal('authModal');
  alert('Добро пожаловать, ' + fio + '!');
}

function renderMiniCalendar() {
  const year = miniCalDate.getFullYear();
  const month = miniCalDate.getMonth();
  const monthNames = ['ЯНВАРЬ','ФЕВРАЛЬ','МАРТ','АПРЕЛЬ','МАЙ','ИЮНЬ','ИЮЛЬ','АВГУСТ','СЕНТЯБРЬ','ОКТЯБРЬ','НОЯБРЬ','ДЕКАБРЬ'];
  document.getElementById('miniCalMonthYear').textContent = monthNames[month] + ' ' + year;
  const grid = document.getElementById('miniCalGrid');
  grid.innerHTML = '';
  dayNamesShort.forEach(d => {
    const el = document.createElement('div');
    el.className = 'day-header';
    el.textContent = d;
    grid.appendChild(el);
  });
  const firstDay = new Date(year, month, 1);
  let startDay = firstDay.getDay();
  startDay = startDay === 0 ? 6 : startDay - 1;
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const daysInPrevMonth = new Date(year, month, 0).getDate();
  for (let i = startDay - 1; i >= 0; i--) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = daysInPrevMonth - i;
    grid.appendChild(el);
  }
  for (let d = 1; d <= daysInMonth; d++) {
    const el = document.createElement('div');
    el.className = 'day-cell';
    if (d === miniCalSelectedDay) el.classList.add('selected');
    el.textContent = d;
    el.onclick = () => {
      miniCalSelectedDay = d;
      document.querySelectorAll('.mini-calendar-grid .day-cell').forEach(c => c.classList.remove('selected'));
      el.classList.add('selected');
      document.getElementById('rescheduleDate').value = year + '-' + String(month + 1).padStart(2, '0') + '-' + String(d).padStart(2, '0');
    };
    grid.appendChild(el);
  }
  const totalCells = startDay + daysInMonth;
  const remaining = (7 - totalCells % 7) % 7;
  for (let i = 1; i <= remaining; i++) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = i;
    grid.appendChild(el);
  }
}
function changeMonth(offset) {
  miniCalDate.setMonth(miniCalDate.getMonth() + offset);
  miniCalSelectedDay = 1;
  renderMiniCalendar();
}
