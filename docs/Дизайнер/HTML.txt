let weekStart = new Date(2026, 9, 1);
function getMonday(d) {
  const date = new Date(d);
  const day = date.getDay();
  const diff = date.getDate() - day + (day === 0 ? -6 : 1);
  return new Date(date.setDate(diff));
}
weekStart = getMonday(new Date(2026, 9, 1));
let currentDayIndex = 0;
let selectedRoomIndex = 0;
let selectedSlot = null;
let miniCalDate = new Date(2026, 10, 1);
let miniCalSelectedDay = 8;

// ===== АВТОРИЗАЦИЯ =====
let currentUser = null;
function getUsers() {
  try { return JSON.parse(localStorage.getItem('roomBookingUsers') || '[]'); }
  catch { return []; }
}
function saveUsers(users) { localStorage.setItem('roomBookingUsers', JSON.stringify(users)); }
function findUser(fio, phone) {
  const users = getUsers();
  const fioNorm = fio.trim().toLowerCase();
  const phoneNorm = phone.replace(/\D/g, '');
  return users.find(u => u.fio.toLowerCase() === fioNorm && u.phone.replace(/\D/g, '') === phoneNorm);
}
function updateAuthButton() {
  const btn = document.getElementById('authBtn');
  if (currentUser) {
    btn.textContent = currentUser.fio;
    btn.classList.add('logged-in');
    btn.onclick = () => {
      if (confirm('Выйти из аккаунта?')) {
        currentUser = null;
        updateAuthButton();
      }
    };
  } else {
    btn.textContent = 'Войти / Зарегистрироваться';
    btn.classList.remove('logged-in');
    btn.onclick = openAuthModal;
  }
}
function openAuthModal() {
  document.getElementById('authFio').value = '';
  document.getElementById('authPhone').value = '';
  document.getElementById('authModal').classList.add('active');
}
function validateAuthFields() {
  const fio = document.getElementById('authFio').value.trim();
  const phone = document.getElementById('authPhone').value.trim();
  const fioValid = fio.split(/\s+/).length >= 2 && fio.length >= 5;
  const phoneDigits = phone.replace(/\D/g, '');
  const phoneValid = phoneDigits.length >= 10;
  return { fio, phone, fioValid, phoneValid, valid: fioValid && phoneValid };
}
function showWarning(text) {
  document.getElementById('warningText').textContent = text;
  document.getElementById('warningModal').classList.add('active');
}
function handleRegister() {
  const { fio, phone, valid } = validateAuthFields();
  if (!valid) {
    showWarning('Вы ввели не все данные или они не корректны');
    return;
  }
  const existing = findUser(fio, phone);
  if (existing) {
    showWarning('Вы уже зарегистрированы! Пожалуйста, нажмите кнопку Войти');
    return;
  }
  const users = getUsers();
  users.push({ fio, phone });
  saveUsers(users);
  currentUser = { fio, phone };
  updateAuthButton();
  closeModal('authModal');
  alert('Регистрация успешна! Добро пожаловать, ' + fio);
}
function handleLogin() {
  const { fio, phone, valid } = validateAuthFields();
  if (!valid) {
    showWarning('Вы ввели не все данные или они не корректны');
    return;
  }
  const existing = findUser(fio, phone);
  if (!existing) {
    showWarning('Вы еще не зарегистрированы! Пожалуйста, нажмите кнопку Зарегистрироваться');
    return;
  }
  currentUser = { fio, phone };
  updateAuthButton();
  closeModal('authModal');
  alert('Добро пожаловать, ' + fio + '!');
}

// ===== НАВИГАЦИЯ =====
function showPage(pageName, btn) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
  document.getElementById('page-' + pageName).classList.add('active');
  if (btn) btn.classList.add('active');
  if (pageName === 'my-bookings') renderMyBookings();
  window.scrollTo(0, 0);
}

// ===== НЕДЕЛЬНЫЙ КАЛЕНДАРЬ =====
function updateWeekCalendar() {
  for (let i = 0; i < 6; i++) {
    const d = new Date(weekStart);
    d.setDate(d.getDate() + i);
    document.getElementById('wd-' + i).textContent =
      d.getDate() + ' ' + monthNamesGenitive[d.getMonth()].substring(0, 3).toUpperCase();
  }
}
function selectDay(index) {
  currentDayIndex = index;
  document.querySelectorAll('.week-day').forEach((el, i) => el.classList.toggle('active', i === index));
}

// ===== ВЫБОР ПОМЕЩЕНИЯ И СЛОТА =====
function selectBookingRoom(index) {
  selectedRoomIndex = index;
  document.querySelectorAll('.room-list-item').forEach((el, i) => el.classList.toggle('active', i === index));
  selectedSlot = null;
  document.querySelectorAll('.time-slot').forEach(s => s.classList.remove('selected'));
}
function selectSlot(el) {
  document.querySelectorAll('.time-slot').forEach(s => s.classList.remove('selected'));
  el.classList.add('selected');
  selectedSlot = el.textContent.trim();
}

// ===== ОБРАБОТЧИК КНОПКИ "БРОНИРОВАТЬ" =====
function handleBookingClick() {
  if (!currentUser) {
    openAuthModal();
  } else {
    openBookingModal();
  }
}

// ===== МОДАЛКА БРОНИРОВАНИЯ =====
function openBookingModal() {
  const room = roomNames[selectedRoomIndex];
  const time = selectedSlot || 'не выбрано';
  document.getElementById('bookingModalTitle').textContent =
    'Забронировать ' + room + ' на ' + time;
  document.getElementById('bookingModal').classList.add('active');
}
function submitBooking() {
  myBookings.push({
    roomIndex: selectedRoomIndex,
    roomName: roomNames[selectedRoomIndex],
    roomImage: roomImages[selectedRoomIndex],
    time: selectedSlot || 'не выбрано',
    name: currentUser.fio,
    phone: currentUser.phone,
    date: new Date().toLocaleDateString('ru-RU')
  });
  closeModal('bookingModal');
  selectedSlot = null;
  document.querySelectorAll('.time-slot').forEach(s => s.classList.remove('selected'));
  alert('Бронь оформлена!\nПомещение: ' + roomNames[selectedRoomIndex] + '\nВремя: ' + (selectedSlot || 'не выбрано'));
}

// ===== РЕНДЕР МОИХ БРОНЕЙ =====
function renderMyBookings() {
  const list = document.getElementById('bookingsList');
  const actions = document.getElementById('bookingActions');
  if (myBookings.length === 0) {
    list.innerHTML = '<div class="no-bookings">У вас пока нет активных бронирований</div>';
    actions.style.display = 'none';
    return;
  }
  list.innerHTML = '';
  myBookings.forEach((booking, index) => {
    const card = document.createElement('div');
    card.className = 'booking-card';
    card.onclick = () => selectBooking(index);
    card.innerHTML = `
      <img src="${booking.roomImage}" alt="${booking.roomName}" class="booking-card-img">
      <div class="booking-card-info">
        <h3>${booking.roomName} на ${booking.time}</h3>
        <p>📅 ${booking.date}</p>
        <p> ${booking.name}</p>
        <p>📞 ${booking.phone}</p>
      </div>
    `;
    list.appendChild(card);
  });
}
function selectBooking(index) {
  selectedBookingIndex = index;
  document.getElementById('bookingActions').style.display = 'flex';
  document.querySelectorAll('.booking-card').forEach((card, i) => {
    card.style.borderColor = i === index ? '#2563EB' : '#E0E0E0';
  });
}

// ===== ОТМЕНА И ПЕРЕНОС =====
function openCancelModalFromMyBookings() {
  if (selectedBookingIndex === null) { alert('Пожалуйста, выберите бронь'); return; }
  const b = myBookings[selectedBookingIndex];
  document.getElementById('cancelModalText').textContent =
    'Вы точно хотите отменить бронь ' + b.roomName + ' на ' + b.time + '?';
  document.getElementById('cancelModal').classList.add('active');
}
function confirmCancel() {
  if (selectedBookingIndex !== null) {
    myBookings.splice(selectedBookingIndex, 1);
    selectedBookingIndex = null;
    renderMyBookings();
  }
  closeModal('cancelModal');
}
function openRescheduleModalFromMyBookings() {
  if (selectedBookingIndex === null) { alert('Пожалуйста, выберите бронь'); return; }
  renderMiniCalendar();
  document.getElementById('rescheduleModal').classList.add('active');
}
function confirmReschedule() {
  const date = document.getElementById('rescheduleDate').value;
  const time = document.getElementById('rescheduleTime').value;
  if (!date || !time) { alert('Пожалуйста, выберите дату и время'); return; }
  if (selectedBookingIndex !== null) {
    myBookings[selectedBookingIndex].time = time;
    myBookings[selectedBookingIndex].date = new Date(date).toLocaleDateString('ru-RU');
    selectedBookingIndex = null;
    renderMyBookings();
  }
  closeModal('rescheduleModal');
}

// ===== МИНИ-КАЛЕНДАРЬ =====
function renderMiniCalendar() {
  const year = miniCalDate.getFullYear();
  const month = miniCalDate.getMonth();
  document.getElementById('miniCalMonthYear').textContent = monthNames[month] + ' ' + year;
  const grid = document.getElementById('miniCalGrid');
  grid.innerHTML = '';
  dayNamesShort.forEach(d => {
    const el = document.createElement('div');
    el.className = 'day-header';
    el.textContent = d;
    grid.appendChild(el);
  });
  const firstDay = new Date(year, month, 1);
  let startDay = firstDay.getDay();
  startDay = startDay === 0 ? 6 : startDay - 1;
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const daysInPrevMonth = new Date(year, month, 0).getDate();
  for (let i = startDay - 1; i >= 0; i--) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = daysInPrevMonth - i;
    grid.appendChild(el);
  }
  for (let d = 1; d <= daysInMonth; d++) {
    const el = document.createElement('div');
    el.className = 'day-cell';
    if (d === miniCalSelectedDay) el.classList.add('selected');
    el.textContent = d;
    el.onclick = () => {
      miniCalSelectedDay = d;
      document.querySelectorAll('.mini-calendar-grid .day-cell').forEach(c => c.classList.remove('selected'));
      el.classList.add('selected');
      document.getElementById('rescheduleDate').value =
        year + '-' + String(month + 1).padStart(2, '0') + '-' + String(d).padStart(2, '0');
    };
    grid.appendChild(el);
  }
  const totalCells = startDay + daysInMonth;
  const remaining = (7 - totalCells % 7) % 7;
  for (let i = 1; i <= remaining; i++) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = i;
    grid.appendChild(el);
  }
}
function changeMonth(offset) {
  miniCalDate.setMonth(miniCalDate.getMonth() + offset);
  miniCalSelectedDay = 1;
  renderMiniCalendar();
}
