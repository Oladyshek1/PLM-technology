  }

function formatPhoneValue(rawVal) {
  if (!rawVal) return '';
  if (rawVal.trim() === '+') return '+7 (';
  const rawDigits = rawVal.replace(/\D/g, '');
  if (!rawDigits) return '';
  let subDigits = '';
  if (rawDigits[0] === '7' || rawDigits[0] === '8') {
    if (rawDigits.length === 1) return '+7 (';
    subDigits = rawDigits.substring(1, 11);
  } else {
    subDigits = rawDigits.substring(0, 10);
  }
  if (!subDigits) return '+7 (';
  let res = '+7 (';
  res += subDigits.substring(0, Math.min(3, subDigits.length));
  if (subDigits.length > 3) res += ') ' + subDigits.substring(3, Math.min(6, subDigits.length));
  if (subDigits.length > 6) res += '-' + subDigits.substring(6, Math.min(8, subDigits.length));
  if (subDigits.length > 8) res += '-' + subDigits.substring(8, 10);
  return res;
}
function getCursorPositionForDigitCount(formatted, digitCount) {
  if (!formatted) return 0;
  if (digitCount <= 0) return formatted.indexOf('(') !== -1 ? formatted.indexOf('(') + 1 : formatted.length;
  let count = 0;
  for (let i = 0; i < formatted.length; i++) {
    if (/\d/.test(formatted[i])) {
      if (i > 1) {
        count++;
        if (count === digitCount) {
          let nextPos = i + 1;
          if (formatted[nextPos] === ')') nextPos += 2;
          else if (formatted[nextPos] === '-') nextPos += 1;
          return Math.min(nextPos, formatted.length);
        }
      }
    }
  }
  return formatted.length;
}
function handlePhoneInput(input) {
  const rawVal = input.value;
  if (!rawVal) { input.value = ''; return; }
  const selStart = input.selectionStart !== null ? input.selectionStart : rawVal.length;
  const textBefore = rawVal.substring(0, selStart);
  let digitsBefore = textBefore.replace(/\D/g, '');
  let subDigitsBefore = 0;
  if (digitsBefore.startsWith('7') || digitsBefore.startsWith('8')) {
    subDigitsBefore = digitsBefore.substring(1).length;
  } else {
    subDigitsBefore = digitsBefore.length;
  }
  const formatted = formatPhoneValue(rawVal);
  input.value = formatted;
  const newPos = getCursorPositionForDigitCount(formatted, subDigitsBefore);
  input.setSelectionRange(newPos, newPos);
}
function handlePhoneKeyDown(e) {
  const input = e.target;
  if (e.key === 'Backspace') {
    const selStart = input.selectionStart;
    const selEnd = input.selectionEnd;
    if ((selStart === 0 && selEnd === input.value.length) || input.value.length <= 4) {
      input.value = '';
      e.preventDefault();
      return;
    }
    if (selStart === selEnd) {
      const textBefore = input.value.substring(0, selStart);
      const digitsBefore = textBefore.replace(/\D/g, '');
      const subDigitsBefore = (digitsBefore.startsWith('7') || digitsBefore.startsWith('8'))
        ? digitsBefore.substring(1).length : digitsBefore.length;
      if (selStart <= 4 && subDigitsBefore === 0) {
        input.value = '';
        e.preventDefault();
        return;
      }
      const charBefore = input.value[selStart - 1];
      if (!/\d/.test(charBefore)) {
        e.preventDefault();
        let deleteIdx = -1;
        for (let i = selStart - 1; i >= 0; i--) {
          if (/\d/.test(input.value[i])) {
            if (i > 1) { deleteIdx = i; }
            break;
          }
        }
        if (deleteIdx !== -1) {
          const val = input.value;
          const newVal = val.substring(0, deleteIdx) + val.substring(deleteIdx + 1);
          const formatted = formatPhoneValue(newVal);
          input.value = formatted;
          const targetSubCount = Math.max(0, subDigitsBefore - 1);
          const newPos = getCursorPositionForDigitCount(formatted, targetSubCount);
          input.setSelectionRange(newPos, newPos);
        } else { input.value = ''; }
      }
    }
  } else if (e.key === 'Delete') {
    const selStart = input.selectionStart;
    const selEnd = input.selectionEnd;
    if (selStart === selEnd && selStart < input.value.length) {
      const charAfter = input.value[selStart];
      if (!/\d/.test(charAfter)) {
        e.preventDefault();
        let deleteIdx = -1;
        for (let i = selStart; i < input.value.length; i++) {
          if (/\d/.test(input.value[i])) { deleteIdx = i; break; }
        }
        if (deleteIdx !== -1) {
          const val = input.value;
          const newVal = val.substring(0, deleteIdx) + val.substring(deleteIdx + 1);
          const formatted = formatPhoneValue(newVal);
          input.value = formatted;
          const textBefore = val.substring(0, selStart);
          const digitsBefore = textBefore.replace(/\D/g, '');
          const subDigitsBefore = (digitsBefore.startsWith('7') || digitsBefore.startsWith('8'))
            ? digitsBefore.substring(1).length : digitsBefore.length;
          const newPos = getCursorPositionForDigitCount(formatted, subDigitsBefore);
          input.setSelectionRange(newPos, newPos);
        }
      }
    }
  }
}

function validateAuthFields() {
  const fio = document.getElementById('authFio').value.trim();
  const phoneNormalized = getNormalizedPhone();
  const fioLettersOnly = /^[a-zA-Zа-яА-ЯёЁ\s\-]+$/.test(fio);
  const fioLengthValid = fio.length >= 2;
  const fioValid = fioLengthValid && fioLettersOnly;
  const phoneValid = phoneNormalized.length === 12;
  let errorMessage = '';
  if (!fio || !phoneNormalized) errorMessage = 'Вы ввели не все данные';
  else if (!fioLettersOnly) errorMessage = 'В ФИО допустимы только буквы кириллицы и латиницы (без цифр и спецсимволов)';
  else if (!fioLengthValid) errorMessage = 'ФИО должно содержать не менее 2 символов';
  else if (!phoneValid) errorMessage = 'Введите корректный номер телефона';
  return { fio, phone: phoneNormalized, fioValid, phoneValid, valid: fioValid && phoneValid, errorMessage };
}
async function handleRegister() {
  clearAuthError();
  const { fio, phone, valid, errorMessage } = validateAuthFields();
  if (!valid) { showAuthError(errorMessage || 'Вы ввели не все данные или они некорректны'); return; }
  const res = await apiFetch('/users', { method: 'POST', body: JSON.stringify({ name: fio, phone: phone }) });
  if (res.isNetworkError) { showAuthError('Сервер БД недоступен. Проверьте запуск uvicorn app:app'); return; }
  if (res.status === 409) { showAuthError('Аккаунт с таким телефоном уже существует! Нажмите «Войти»'); return; }
  if (!res.ok) { showAuthError(res.data?.detail || 'Ошибка регистрации на сервере'); return; }
  saveStoredUser(res.data);
  updateAuthButton();
  closeModal('authModal');
  showToast('Регистрация успешна! Добро пожаловать, ' + (res.data.name || fio), 'success');
  loadScheduleGrid();
}
async function handleLogin() {
  clearAuthError();
  const { fio, phone, valid, errorMessage } = validateAuthFields();
  if (!valid) { showAuthError(errorMessage || 'Вы ввели не все данные или они некорректны'); return; }
  const queryParams = new URLSearchParams({ phone: phone });
  if (fio) queryParams.append('name', fio);
  const res = await apiFetch(`/users?${queryParams.toString()}`);
  if (res.isNetworkError) { showAuthError('Сервер БД недоступен. Проверьте запуск uvicorn app:app'); return; }
  if (!res.ok) { showAuthError(res.data?.detail || 'Ошибка при входе'); return; }
  const users = res.data || [];
  if (users.length === 0) { showAuthError('Аккаунт не найден! Пожалуйста, нажмите «Зарегистрироваться»'); return; }
  saveStoredUser(users[0]);
  updateAuthButton();
  closeModal('authModal');
  showToast('Добро пожаловать, ' + (users[0].name || fio) + '!', 'success');
  loadScheduleGrid();
}

function applyFilters() {
  const roomId = document.getElementById('filterRoom').value;
  const dateFrom = document.getElementById('filterDateFrom').value;
  const dateTo = document.getElementById('filterDateTo').value;

  if (dateFrom && dateTo && dateFrom > dateTo) {
    showToast('Дата «С» не может быть позже даты «По»', 'error');
    return;
  }

  let filtered = allMyBookings.slice();

  if (roomId) {
    filtered = filtered.filter(b => String(b.resource_id) === String(roomId));
  }
  if (dateFrom) {
    filtered = filtered.filter(b => {
      const d = b.date || b.booking_date;
      return d && d >= dateFrom;
    });
  }
  if (dateTo) {
    filtered = filtered.filter(b => {
      const d = b.date || b.booking_date;
      return d && d <= dateTo;
    });
  }

  myBookings = filtered;
  renderMyBookingsList();
  showToast(`Найдено записей: ${filtered.length}`, 'success');
}

function clearSelectedSlot(periodId) {
  const idx = selectedSlots.findIndex(s => s.id === periodId);
  if (idx !== -1) selectedSlots.splice(idx, 1);
  selectedSlot = selectedSlots.length > 0 ? selectedSlots[0] : null;

function updateTimeSlotsWithBookings() {
  const now = new Date();
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const selectedDate = new Date(weekStart);
  selectedDate.setDate(selectedDate.getDate() + currentDayIndex);
  selectedDate.setHours(0, 0, 0, 0);
  const isToday = selectedDate.getTime() === today.getTime();
  const isPastDay = selectedDate < today;
  document.querySelectorAll('.time-slot').forEach(slot => {
    const periodId = parseInt(slot.dataset.periodId, 10);
    const startHour = parseInt(slot.dataset.startHour, 10);
    slot.classList.remove('past-slot', 'occupied-other', 'occupied-mine');
    slot.removeAttribute('title');
    let isPastTime = false;
    if (isPastDay) isPastTime = true;
    else if (isToday) {
      const slotStartTime = new Date(selectedDate);
      slotStartTime.setHours(startHour, 0, 0, 0);
      const cutoffTime = new Date(slotStartTime.getTime() - 2 * 60 * 1000);
      if (now >= cutoffTime) isPastTime = true;
    }
    if (isPastTime) {
      slot.style.display = 'none';
      if (slot.classList.contains('selected')) slot.classList.remove('selected');
      clearSelectedSlot(periodId);
      return;
    }
    const booking = currentDayBookings.find(b => b.time_id === periodId);
    if (booking) {
      slot.style.display = 'none';
      if (slot.classList.contains('selected')) slot.classList.remove('selected');
      clearSelectedSlot(periodId);
    } else {
      slot.style.display = '';
    }
  });
}

function updateWeekCalendar() {
  const container = document.getElementById('weekDaysContainer');
  container.innerHTML = '';
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  let defaultDayToSelect = -1;
  for (let i = 0; i < 7; i++) {
    const d = new Date(weekStart);
    d.setDate(d.getDate() + i);
    d.setHours(0, 0, 0, 0);
    const isPast = d < today;
    const isToday = d.getTime() === today.getTime();
    const dayEl = document.createElement('div');
    dayEl.className = 'week-day' + (isPast ? ' past-day' : '');
    dayEl.dataset.day = i;
    const dayNameText = dayNamesShort[i];
    const dayDateText = d.getDate() + ' ' + monthNamesGenitive[d.getMonth()].substring(0, 3).toUpperCase();
    dayEl.innerHTML = `<div class="day-name">${dayNameText}</div><div class="day-date">${dayDateText}</div>`;
    if (isPast) {
      dayEl.title = 'Этот день уже прошёл';
    } else {
      dayEl.onclick = () => selectDay(i);
      if (defaultDayToSelect === -1) {
        if (isToday || defaultDayToSelect === -1) defaultDayToSelect = i;
      }
    }
    container.appendChild(dayEl);
  }
  const selectedDate = new Date(weekStart);
  selectedDate.setDate(selectedDate.getDate() + currentDayIndex);
  selectedDate.setHours(0, 0, 0, 0);
  if (selectedDate < today || currentDayIndex < 0 || currentDayIndex >= 7) {
    currentDayIndex = defaultDayToSelect !== -1 ? defaultDayToSelect : 0;
  }
  highlightSelectedDay();
  loadScheduleGrid();
}
function highlightSelectedDay() {
  const days = document.querySelectorAll('#weekDaysContainer .week-day');
  days.forEach((el, i) => { el.classList.toggle('active', i === currentDayIndex); });
}
function selectDay(index) {
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const targetDate = new Date(weekStart);
  targetDate.setDate(targetDate.getDate() + index);
  targetDate.setHours(0, 0, 0, 0);
  if (targetDate < today) return;
  currentDayIndex = index;
  highlightSelectedDay();
  selectedSlots = [];
  selectedSlot = null;
  loadScheduleGrid();
}

function renderPickerCalendar() {
  const year = pickerCalDate.getFullYear();
  const month = pickerCalDate.getMonth();
  document.getElementById('pickerCalMonthYear').textContent = monthNames[month] + ' ' + year;
  const grid = document.getElementById('pickerCalGrid');
  grid.innerHTML = '';
  dayNamesShort.forEach(d => {
    const el = document.createElement('div');
    el.className = 'day-header';
    el.textContent = d;
    grid.appendChild(el);
  });
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const firstDay = new Date(year, month, 1);
  let startDay = firstDay.getDay();
  startDay = startDay === 0 ? 6 : startDay - 1;
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const daysInPrevMonth = new Date(year, month, 0).getDate();
  for (let i = startDay - 1; i >= 0; i--) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = daysInPrevMonth - i;
    grid.appendChild(el);
  }
  for (let d = 1; d <= daysInMonth; d++) {
    const cellDate = new Date(year, month, d);
    cellDate.setHours(0, 0, 0, 0);
    const isPast = cellDate < today;
    const isToday = cellDate.getTime() === today.getTime();
    const isSelected = pickerSelectedDate &&
      cellDate.getFullYear() === pickerSelectedDate.getFullYear() &&
      cellDate.getMonth() === pickerSelectedDate.getMonth() &&
      cellDate.getDate() === pickerSelectedDate.getDate();
    const el = document.createElement('div');
    el.className = 'day-cell' + (isPast ? ' past-mini-day' : '') + (isToday ? ' today' : '') + (isSelected ? ' selected' : '');
    el.textContent = d;
    if (!isPast) {
      el.onclick = () => {
        pickerSelectedDate = new Date(year, month, d);
        pickerSelectedDate.setHours(0, 0, 0, 0);
        document.querySelectorAll('#pickerCalGrid .day-cell').forEach(c => c.classList.remove('selected'));
        el.classList.add('selected');
      };
      el.ondblclick = () => { pickerSelectedDate = new Date(year, month, d); confirmPickerDate(); };
    }
    grid.appendChild(el);
  }
  const totalCells = startDay + daysInMonth;
  const remaining = (7 - totalCells % 7) % 7;
  for (let i = 1; i <= remaining; i++) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = i;
    grid.appendChild(el);
  }
}
function changePickerMonth(offset) {
  pickerCalDate.setMonth(pickerCalDate.getMonth() + offset);
  renderPickerCalendar();
}
function confirmPickerDate() {
  if (!pickerSelectedDate) { showToast('Пожалуйста, выберите дату', 'error'); return; }
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  if (pickerSelectedDate < today) { showToast('Нельзя выбирать прошедшие даты', 'error'); return; }
  weekStart = getMonday(pickerSelectedDate);
  const day = pickerSelectedDate.getDay();
  currentDayIndex = day === 0 ? 6 : day - 1;
  updateWeekCalendar();
  closeModal('calendarPickerModal');
  showToast('Переход к дате: ' + pickerSelectedDate.toLocaleDateString('ru-RU'), 'success');
}

function selectBookingRoom(index) {
  selectedRoomIndex = index;

function selectSlot(el, periodObj) {
  if (el.classList.contains('past-slot') || el.style.display === 'none') return;
  const periodId = periodObj ? periodObj.id : parseInt(el.dataset.periodId, 10);
  const timeLabel = el.dataset.time || el.textContent.trim();
  const startHour = parseInt(el.dataset.startHour, 10);
  const existingIndex = selectedSlots.findIndex(s => s.id === periodId);
  if (existingIndex !== -1) {
    selectedSlots.splice(existingIndex, 1);
    el.classList.remove('selected');
  } else {
    selectedSlots.push({ id: periodId, time: timeLabel, period: periodObj, startHour: startHour });
    el.classList.add('selected');
  }
  selectedSlots.sort((a, b) => a.startHour - b.startHour);
  selectedSlot = selectedSlots.length > 0 ? selectedSlots[0] : null;
  if (selectedSlots.length > 0) document.getElementById('bookingHint').classList.remove('active');
}

function openBookingModal() {

  const targetDate = new Date(weekStart);
  targetDate.setDate(targetDate.getDate() + currentDayIndex);
  let timeSummary = '';
  if (selectedSlots.length === 1) timeSummary = selectedSlots[0].time;
  else if (selectedSlots.length > 1) timeSummary = selectedSlots.map(s => s.time).join(', ');
  else timeSummary = 'не выбрано';
  document.getElementById('bookingModalTitle').textContent =
    'Забронировать ' + room + ' на ' + targetDate.toLocaleDateString('ru-RU') + ' (' + timeSummary + ')';
  document.getElementById('bookingModal').classList.add('active');
}
async function submitBooking() {
  if (!currentUser) { closeModal('bookingModal'); openAuthModal(); return; }
  if (selectedSlots.length === 0) { closeModal('bookingModal'); return; }
  const room = roomsData[selectedRoomIndex] || { id: selectedRoomIndex + 1, name: fallbackRoomNames[selectedRoomIndex] };
  const dateISO = getSelectedDateISO();
  const timeIds = selectedSlots.map(s => s.id);
  const body = { user_id: currentUser.id, resource_id: room.id, booking_date: dateISO, time_ids: timeIds };
  const res = await apiFetch('/bookings', { method: 'POST', body: JSON.stringify(body) });
  if (res.isNetworkError) { showToast('Сервер БД недоступен. Проверьте backend', 'error'); return; }
  if (res.status === 409) { showToast(res.data?.detail || 'Накладка: выбранное время уже успели забронировать!', 'error'); closeModal('bookingModal'); loadScheduleGrid(); return; }
  if (res.status === 400) { showToast(res.data?.detail || 'Нельзя бронировать прошедшее время!', 'error'); closeModal('bookingModal'); loadScheduleGrid(); return; }
  if (!res.ok) { showToast(res.data?.detail || 'Ошибка создания брони', 'error'); closeModal('bookingModal'); return; }
  closeModal('bookingModal');
  selectedSlots = [];
  selectedSlot = null;
  document.querySelectorAll('.time-slot').forEach(s => s.classList.remove('selected'));
  let bookingIdText = '';
  if (res.data) {
    if (res.data.ids && res.data.ids.length > 1) bookingIdText = ` (ID броней: #${res.data.ids.join(', #')})`;
    else if (res.data.id) bookingIdText = ` (ID брони: #${res.data.id})`;
  }
  showToast(`Бронь успешно оформлена!${bookingIdText}`, 'success');
  loadScheduleGrid();
}

async function renderMyBookings() {
  const list = document.getElementById('bookingsList');
  if (!currentUser) {
    list.innerHTML = '<div class="no-bookings">Войдите в аккаунт, чтобы просмотреть свои бронирования</div>';
    selectedBookingIndex = null;
    allMyBookings = [];
    myBookings = [];
    return;
  }
  list.innerHTML = '<div class="no-bookings">Загрузка броней из базы данных...</div>';
  const res = await apiFetch(`/bookings?user_id=${currentUser.id}`);
  if (res.isNetworkError) {
    list.innerHTML = '<div class="no-bookings">Сервер недоступен. Не удалось загрузить брони</div>';
    return;
  }
  if (!res.ok || !Array.isArray(res.data)) {
    list.innerHTML = '<div class="no-bookings">Ошибка загрузки броней. Попробуйте позже</div>';
    return;
  }
  const activeBookings = res.data.filter(b => b.status !== 'CANCELLED');
  allMyBookings = activeBookings;
  myBookings = activeBookings.slice();
  renderMyBookingsList();
}

function renderMyBookingsList() {
  const list = document.getElementById('bookingsList');
  selectedBookingIndex = null;

  if (myBookings.length === 0) {
    const hasFilters = document.getElementById('filterRoom').value ||
                       document.getElementById('filterDateFrom').value ||
                       document.getElementById('filterDateTo').value;
    list.innerHTML = hasFilters
      ? '<div class="no-bookings">По выбранным фильтрам ничего не найдено. Попробуйте изменить параметры или сбросить фильтры.</div>'
      : '<div class="no-bookings">У вас пока нет активных бронирований</div>';
    return;
  }

  list.innerHTML = '';
  myBookings.forEach((b, index) => {
    const row = document.createElement('div');
    row.className = 'booking-row';
    row.id = `bookingRow-${index}`;

    const imgUrl = getRoomImageUrl(b.resource_id, b.resource);
    const timeLabel = (b.start_time && b.end_time)
      ? `${b.start_time.substring(0,5)} - ${b.end_time.substring(0,5)}`
      : '';
    const dateFormatted = b.date ? new Date(b.date).toLocaleDateString('ru-RU') : (b.booking_date || '');
    const roomMeta = roomsData.find(r => r.id === b.resource_id || r.name === b.resource) || {};
    const areaText = roomMeta.area ? `${roomMeta.area} м²` : '—';
    const capacityText = roomMeta.capacity ? `до ${roomMeta.capacity} чел.` : '—';
    const equipmentText = roomMeta.equipment || 'Стандартное оснащение';

    row.innerHTML = `
      <div class="booking-card" onclick="selectBooking(${index})">
        <img src="${imgUrl}" alt="${b.resource || ''}" class="booking-card-img" onerror="this.onerror=null; this.src='src/images/rooms/assembly-hall.png'">
        <div class="booking-card-info">
          <h3>${b.resource || 'Помещение'} на ${dateFormatted} (${timeLabel})</h3>
          <p>Вместимость: ${capacityText}</p>
          <p>Площадь: ${areaText}</p>
          <p>Оборудование: ${equipmentText}</p>
        </div>
      </div>
    `;
    list.appendChild(row);
  });
}

function selectBooking(index) {
  if (selectedBookingIndex === index) {
    selectedBookingIndex = null;
  } else {
    selectedBookingIndex = index;
  }
  document.querySelectorAll('#bookingsList .booking-row').forEach((row, idx) => {
    const card = row.querySelector('.booking-card');
    const existingActions = row.querySelector('.booking-item-actions');
    const isSelected = selectedBookingIndex === idx;
    if (card) card.classList.toggle('selected', isSelected);
    if (isSelected) {
      if (!existingActions) {
        const actionsDiv = document.createElement('div');
        actionsDiv.className = 'booking-item-actions';
        actionsDiv.innerHTML = `
          <button class="booking-action-btn" onclick="openCancelModalFromMyBookings()">Отменить бронь</button>
          <button class="booking-action-btn" onclick="openRescheduleModalFromMyBookings()">Перенести бронь</button>
        `;
        row.appendChild(actionsDiv);
      }
    } else if (existingActions) {
      existingActions.remove();
    }
  });
}

function openCancelModalFromMyBookings() {

  const timeLabel = (b.start_time && b.end_time)
    ? `${b.start_time.substring(0,5)} - ${b.end_time.substring(0,5)}`
    : '';
  const dateFormatted = b.date ? new Date(b.date).toLocaleDateString('ru-RU') : (b.booking_date || '');
  document.getElementById('cancelModalText').textContent =
    `Вы точно хотите отменить бронь «${b.resource || ''}» на ${dateFormatted} (${timeLabel})?`;

async function confirmCancel() {
  const modal = document.getElementById('cancelModal');
  const bookingId = modal.dataset.bookingId;
  if (!bookingId) { closeModal('cancelModal'); return; }

  const res = await apiFetch(`/bookings/${bookingId}/cancel?user_id=${currentUser.id}`, { method: 'POST' });
  if (res.isNetworkError) { showToast('Сервер БД недоступен', 'error'); closeModal('cancelModal'); return; }

  modal.dataset.bookingId = '';
  selectedBookingIndex = null;
  closeModal('cancelModal');
  showToast('Бронь успешно отменена', 'success');
  if (document.getElementById('page-my-bookings').classList.contains('active')) {
    renderMyBookings();
  }
  loadScheduleGrid();
}

function openRescheduleModalFromMyBookings() {
  if (selectedBookingIndex === null) { showToast('Пожалуйста, выберите бронь из списка', 'error'); return; }
  clearRescheduleError();
  const curBooking = myBookings[selectedBookingIndex];
  const initialDate = curBooking && curBooking.date ? toDateISO(new Date(curBooking.date)) : toDateISO(new Date());
  document.getElementById('rescheduleDate').min = toDateISO(new Date());
  document.getElementById('rescheduleDate').value = initialDate;
  document.getElementById('rescheduleDate').onchange = updateRescheduleAvailableTimes;
  renderRescheduleMiniCalendar();
  updateRescheduleAvailableTimes();
  document.getElementById('rescheduleModal').classList.add('active');

async function updateRescheduleAvailableTimes() {
  const select = document.getElementById('rescheduleTime');
  if (!select) return;
  if (selectedBookingIndex === null || !myBookings[selectedBookingIndex]) {
    select.innerHTML = '<option value="">Выберите время</option>';
    return;
  }
  const curBooking = myBookings[selectedBookingIndex];
  let resId = curBooking.resource_id;
  if (!resId && curBooking.resource && roomsData.length > 0) {
    const found = roomsData.find(r => r.name === curBooking.resource);
    if (found) resId = found.id;
  }
  if (!resId) { select.innerHTML = '<option value="">Ошибка: помещение не определено</option>'; return; }
  const dateVal = document.getElementById('rescheduleDate').value;
  if (!dateVal) { select.innerHTML = '<option value="">Сначала выберите дату</option>'; return; }
  select.innerHTML = '<option value="">Загрузка доступного времени...</option>';
  const res = await apiFetch(`/availability?resource_id=${resId}&booking_date=${dateVal}`);
  select.innerHTML = '<option value="">Выберите время</option>';
  if (!res.ok || !Array.isArray(res.data) || res.data.length === 0) {
    const opt = document.createElement('option');
    opt.value = '';
    opt.disabled = true;
    opt.textContent = 'Нет свободного времени на эту дату';
    select.appendChild(opt);
    return;
  }
  res.data.forEach(p => {
    const opt = document.createElement('option');
    opt.value = p.id;
    opt.textContent = formatPeriodLabel(p.start_time, p.end_time);
    select.appendChild(opt);
  });
}
async function confirmReschedule() {
  clearRescheduleError();
  const dateVal = document.getElementById('rescheduleDate').value;
  const timeIdVal = document.getElementById('rescheduleTime').value;
  if (!dateVal || !timeIdVal) { showRescheduleError('Пожалуйста, выберите дату и время'); return; }
  if (selectedBookingIndex === null || !myBookings[selectedBookingIndex]) { closeModal('rescheduleModal'); return; }
  const currentB = myBookings[selectedBookingIndex];
  let resId = currentB.resource_id;
  if (!resId && currentB.resource && roomsData.length > 0) {
    const found = roomsData.find(r => r.name === currentB.resource);
    if (found) resId = found.id;
  }
  if (!resId) { showRescheduleError('Не удалось определить помещение для переноса'); return; }
  const createRes = await apiFetch('/bookings', {
    method: 'POST',
    body: JSON.stringify({ user_id: currentUser.id, resource_id: resId, booking_date: dateVal, time_id: parseInt(timeIdVal, 10) })
  });
  if (createRes.status === 409) { showRescheduleError(createRes.data?.detail || 'Выбранное время уже занято! Выберите другое время или день.'); return; }
  if (createRes.status === 400) { showRescheduleError(createRes.data?.detail || 'Нельзя переносить на прошедшее время!'); return; }
  if (!createRes.ok || !createRes.data) {
    let msg = 'Ошибка создания новой брони';
    if (createRes.data) {
      if (typeof createRes.data.detail === 'string') msg = createRes.data.detail;
      else if (Array.isArray(createRes.data.detail)) msg = createRes.data.detail.map(e => e.msg || JSON.stringify(e)).join(', ');
      else if (typeof createRes.data.detail === 'object') msg = JSON.stringify(createRes.data.detail);
    }
    showRescheduleError(msg);
    return;
  }
  const newBookingId = createRes.data.id;
  const cancelOldRes = await apiFetch(`/bookings/${currentB.id}/cancel?user_id=${currentUser.id}`, { method: 'POST' });
  if (!cancelOldRes.ok) {
    console.error('Ошибка отмены старой брони, выполняется откат новой брони:', cancelOldRes);
    await apiFetch(`/bookings/${newBookingId}/cancel?user_id=${currentUser.id}`, { method: 'POST' });
    showRescheduleError('Не удалось отменить старую бронь. Перенос отменён во избежание дублирования.');
    return;
  }
  selectedBookingIndex = null;
  closeModal('rescheduleModal');
  const newBookingIdText = newBookingId ? ` (Новый ID брони: #${newBookingId})` : '';
  showToast(`Бронь успешно перенесена!${newBookingIdText}`, 'success');
  renderMyBookings();
  loadScheduleGrid();
}
function renderRescheduleMiniCalendar() {
  const year = rescheduleCalDate.getFullYear();
  const month = rescheduleCalDate.getMonth();
  document.getElementById('miniCalMonthYear').textContent = monthNames[month] + ' ' + year;
  const grid = document.getElementById('miniCalGrid');
  grid.innerHTML = '';
  dayNamesShort.forEach(d => {
    const el = document.createElement('div');
    el.className = 'day-header';
    el.textContent = d;
    grid.appendChild(el);
  });
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const firstDay = new Date(year, month, 1);
  let startDay = firstDay.getDay();
  startDay = startDay === 0 ? 6 : startDay - 1;
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const daysInPrevMonth = new Date(year, month, 0).getDate();
  for (let i = startDay - 1; i >= 0; i--) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = daysInPrevMonth - i;
    grid.appendChild(el);
  }
  for (let d = 1; d <= daysInMonth; d++) {
    const cellDate = new Date(year, month, d);
    cellDate.setHours(0, 0, 0, 0);
    const isPast = cellDate < today;
    const isToday = cellDate.getTime() === today.getTime();
    const el = document.createElement('div');
    el.className = 'day-cell' + (isPast ? ' past-mini-day' : '') + (isToday ? ' today' : '');
    if (d === rescheduleSelectedDay && !isPast) el.classList.add('selected');
    el.textContent = d;
    if (!isPast) {
      el.onclick = () => {
        rescheduleSelectedDay = d;
        document.querySelectorAll('#miniCalGrid .day-cell').forEach(c => c.classList.remove('selected'));
        el.classList.add('selected');
        document.getElementById('rescheduleDate').value =
          year + '-' + String(month + 1).padStart(2, '0') + '-' + String(d).padStart(2, '0');
        clearRescheduleError();
        updateRescheduleAvailableTimes();
      };
    }
    grid.appendChild(el);
  }
  const totalCells = startDay + daysInMonth;
  const remaining = (7 - totalCells % 7) % 7;
  for (let i = 1; i <= remaining; i++) {
    const el = document.createElement('div');
    el.className = 'day-cell other-month';
    el.textContent = i;
    grid.appendChild(el);
  }
}
function changeRescheduleMonth(offset) {
  rescheduleCalDate.setMonth(rescheduleCalDate.getMonth() + offset);
  rescheduleSelectedDay = 1;
  renderRescheduleMiniCalendar();
}
